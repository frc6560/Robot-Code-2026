package frc.robot.utility;

import static org.junit.jupiter.api.Assertions.*;
import edu.wpi.first.networktables.NetworkTableInstance;
import org.junit.jupiter.api.Test;

/** Temporary characterization of the existing parser, not a test of correct behavior. */
class LimelightAllocationProbeTest {
    @Test void tinyPacketCanRequestVmLimitArrayThroughPublicParser() {
        double[] packet = new double[11];
        packet[7] = Integer.MAX_VALUE;
        var entry = NetworkTableInstance.getDefault().getTable("allocation-probe")
            .getEntry("botpose_orb_wpiblue");
        entry.setDoubleArray(packet);
        try {
            // Integer.MAX_VALUE array length exceeds the VM array-size limit immediately;
            // this does not fill the heap with an actually enormous array.
            var error = assertThrows(OutOfMemoryError.class,
                () -> LimelightHelpers.getBotPoseEstimate_wpiBlue_MegaTag2("allocation-probe"));
            System.out.println("LIMELIGHT_PROBE length=11 tagCount=" + packet[7] + " result=" + error);
        } finally {
            entry.unpublish();
            entry.close();
        }
    }
    @Test void validHeaderOnlyFrameCurrentlyRetainsReportedCount() {
        double[] packet = new double[11];
        packet[7] = 2;
        var entry = NetworkTableInstance.getDefault().getTable("allocation-probe-valid")
            .getEntry("botpose_orb_wpiblue");
        entry.setDoubleArray(packet);
        try {
            var pose = LimelightHelpers.getBotPoseEstimate_wpiBlue_MegaTag2("allocation-probe-valid");
            assertNotNull(pose);
            assertEquals(2, pose.tagCount);
            assertEquals(2, pose.rawFiducials.length);
            assertNull(pose.rawFiducials[0]);
        } finally {
            entry.unpublish();
            entry.close();
        }
    }
}
